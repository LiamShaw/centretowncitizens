{{/*
  The Cloudflare Pages redirect map.

  Cloudflare Pages reads a file called _redirects from the root of the
  published site. Every URL the old Squarespace site published needs a
  permanent redirect to its new home, so that inbound links from council
  documents, press coverage and social media keep working.

  The map comes from data/redirects.yaml, which the export writes by pairing
  each old Squarespace path with the path its content now lives at. This
  template renders it during the Hugo build, so the redirect map can never
  be left out of a deployment. tools/build.py renames the result from
  _redirects.txt to _redirects, which is the name Cloudflare expects.

  tools/build_redirects.py and tools/check_coverage.py work from the built
  output and catch any page whose published path differs from the recorded
  one.
*/}}
{{- $rows := .Site.Data.redirects -}}
{{- if not $rows -}}
  {{- errorf "data/redirects.yaml is missing or empty. Run tools/export_squarespace.py before building." -}}
{{- end -}}
# Generated by the Hugo build from data/redirects.yaml. Do not edit by hand.
# Every URL the old Squarespace site published, redirected to its new home.
{{ range $rows -}}
{{ .source }}  {{ .target }}  301
{{ end -}}
